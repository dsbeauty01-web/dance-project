// PARSE STOCK & SELLER — Apify version (FINAL) // Apify returns data as a stringified JSON array inside .data // Uses Apify's native inStock and seller fields const raw = $input.first().json.data; const product = $('Loop Each Product').item.json; // Apify returns data as a stringified JSON. Parse it. let items = []; try { if (typeof raw === 'string') { items = JSON.parse(raw); } else if (Array.isArray(raw)) { items = raw; } else if (raw) { items = [raw]; } } catch (e) { items = []; } const item = items[0] || {}; // === Extract fields === const title = item.title || product.Title || 'Amazon Product'; const imageUrl = item.thumbnailImage || (item.highResolutionImages && item.highResolutionImages[0]) || (item.galleryThumbnails && item.galleryThumbnails[0]) || 'https://m.media-amazon.com/images/G/01/x-locale/common/transparent-pixel.gif'; const priceValue = item.price?.value || null; const priceCurrency = item.price?.currency || '$'; const price = priceValue ? `${priceCurrency}${priceValue}` : null; // === Stock detection — Apify gives us the answer directly === const inStock = item.inStock === true; // === Seller detection === let sellerName = item.seller || 'Unknown'; if (sellerName === null || sellerName === 'Unknown') { // No explicit seller. If brand is Amazon, default to Amazon.com. if (item.brand && /amazon/i.test(item.brand)) { sellerName = 'Amazon.com'; } else if (inStock) { // In stock with no seller info → likely Amazon as seller (common pattern) sellerName = 'Amazon.com'; } else { sellerName = 'Unknown'; } } // === Match seller against TargetSeller === const targetSeller = (product.TargetSeller || '').toLowerCase().trim(); const detectedSeller = String(sellerName).toLowerCase().trim(); let sellerMatches = false; if (targetSeller === 'any' || targetSeller === '') { sellerMatches = true; } else if (targetSeller.includes('amazon')) { sellerMatches = /amazon/i.test(detectedSeller); } else { sellerMatches = detectedSeller.includes(targetSeller) || targetSeller.includes(detectedSeller); } // === Cooldown: don't re-alert within 6 hours === const lastAlerted = product.LastAlerted ? new Date(product.LastAlerted) : null; const cooldownMs = 6 * 60 * 60 * 1000; const alreadyAlerted = lastAlerted && (Date.now() - lastAlerted.getTime() < cooldownMs); return [{ json: { ASIN: product.ASIN, rowNumber: product.row_number, title, price, imageUrl, sellerName, targetSeller: product.TargetSeller, inStock, sellerMatches, alreadyAlerted: !!alreadyAlerted, affiliateTag: product.AffiliateTag || '', debugInfo: { priceValue, apifyInStock: item.inStock, apifySeller: item.seller, itemFound: items.length > 0, lastAlerted: product.LastAlerted, } } }];
